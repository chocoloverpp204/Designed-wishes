<script>
/* OPEN THE SURPRISE */
function openSurprise() {
  document.getElementById('birthday').scrollIntoView({
    behavior: 'smooth'
  });
}

/* OPEN AND CLOSE THE SURPRISE CARDS */
function toggleCard(card) {
  const isOpen = card.classList.toggle('open');
  card.setAttribute('aria-expanded', String(isOpen));
}

/* SHOW OR HIDE THE PERSONAL LETTER */
function toggleLetter() {
  const letter = document.getElementById('letterContent');
  const button = document.getElementById('letterButton');

  const isOpen = letter.classList.toggle('show');

  button.textContent = isOpen
    ? 'Close my letter ♡'
    : 'Read my letter ♡';

  button.setAttribute('aria-expanded', String(isOpen));
}

/* CREATE A GENTLE, TWINKLING STAR FIELD */
const canvas = document.getElementById('stars');
const ctx = canvas.getContext('2d');

let stars = [];
let width = 0;
let height = 0;
let animationFrame;

function resizeCanvas() {
  const dpr = Math.min(window.devicePixelRatio || 1, 2);

  width = window.innerWidth;
  height = window.innerHeight;

  canvas.width = Math.floor(width * dpr);
  canvas.height = Math.floor(height * dpr);
  canvas.style.width = width + 'px';
  canvas.style.height = height + 'px';

  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

  const count = Math.min(
    180,
    Math.max(65, Math.floor((width * height) / 8500))
  );

  stars = Array.from({ length: count }, () => ({
    x: Math.random() * width,
    y: Math.random() * height,
    radius: Math.random() * 1.5 + 0.3,
    phase: Math.random() * Math.PI * 2,
    speed: Math.random() * 0.018 + 0.004
  }));
}

function drawStars() {
  ctx.clearRect(0, 0, width, height);

  for (const star of stars) {
    star.phase += star.speed;

    const opacity =
      0.25 + (Math.sin(star.phase) + 1) * 0.3;

    ctx.beginPath();
    ctx.arc(star.x, star.y, star.radius, 0, Math.PI * 2);
    ctx.fillStyle = `rgba(235, 225, 255, ${opacity})`;
    ctx.fill();
  }

  animationFrame = requestAnimationFrame(drawStars);
}

resizeCanvas();
drawStars();

window.addEventListener('resize', resizeCanvas);

window.addEventListener('pagehide', () => {
  cancelAnimationFrame(animationFrame);
});
